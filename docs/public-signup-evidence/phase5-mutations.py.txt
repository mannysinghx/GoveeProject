only = sys.argv[1:]
results = []
for name, rel, old, new, test, pattern in MUTATIONS:
    if only and not any(o in name for o in only):
        continue
    path = S + rel
    original = open(path).read()
    if original.count(old) != 1:
        results.append(f'NOT APPLIED ({original.count(old)} matches): {name}')
        continue
    open(path, 'w').write(original.replace(old, new))
    try:
        run = subprocess.run(['npx', 'vitest', 'run', test, '-t', pattern], cwd=S, capture_output=True, text=True, timeout=300)
        out = run.stdout + run.stderr
        ran = ' passed' in out or ' failed' in out
        verdict = 'CAUGHT' if run.returncode != 0 and 'failed' in out else ('NO TESTS RAN' if not ran else 'MISSED')
        results.append(f'{verdict}: {name}')
    finally:
        open(path, 'w').write(original)
print('\n'.join(results))
