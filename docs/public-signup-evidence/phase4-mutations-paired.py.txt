import subprocess
S='/app/apps/server/'
p=S+'src/auth/signup-service.ts'
orig=open(p).read()
def run(name, edits, test, pattern):
    s=orig
    for old,new in edits:
        assert s.count(old)==1, (name, old[:50], s.count(old))
        s=s.replace(old,new)
    open(p,'w').write(s)
    try:
        r=subprocess.run(['npx','vitest','run',test,'-t',pattern],cwd=S,capture_output=True,text=True)
        out=r.stdout+r.stderr
        print(('CAUGHT' if r.returncode!=0 and 'failed' in out else 'MISSED')+': '+name)
    finally:
        open(p,'w').write(orig)
