// Does the normal "rebuild a table" recipe survive this repo's migration runner? Throwaway data only.
import fs from 'node:fs'; import os from 'node:os'; import path from 'node:path';
import { openDatabase, runMigrations } from './dist/database/database.js';
const REAL = '/tmp/app/migrations';
function seed(db, withPinnedPreset) {
  const now = new Date().toISOString();
  db.prepare(`INSERT INTO devices (id,govee_device_id,sku,name,capabilities_json,discovery_status,first_discovered_at,last_discovered_at,created_at,updated_at) VALUES ('d1','g1','H805C','L','[]','present',?,?,?,?)`).run(now,now,now,now);
  db.prepare(`INSERT INTO device_states (device_id,reported_state_json,reported_at,source) VALUES ('d1','{}',?,'poll')`).run(now);
  db.prepare(`INSERT INTO capability_states (device_id,instance) VALUES ('d1','powerSwitch')`).run();
  db.prepare(`INSERT INTO groups (id,name,created_at,updated_at) VALUES ('g','G',?,?)`).run(now,now);
  db.prepare(`INSERT INTO group_devices VALUES ('g','d1')`).run();
  db.prepare(`INSERT INTO command_history (id,device_id,source_type,action_type,instance,request_id,request_summary,status,created_at) VALUES ('c','d1','manual','power','powerSwitch','r','s','accepted',?)`).run(now);
  db.prepare(`INSERT INTO activity_events (created_at,event_type,outcome,device_id,message) VALUES (?,'x','info','d1','m')`).run(now);
  if (withPinnedPreset) {
    db.prepare(`INSERT INTO presets (id,name,created_at,updated_at) VALUES ('p','P',?,?)`).run(now,now);
    db.prepare(`INSERT INTO preset_actions (preset_id,device_id,action_type,action_payload_json,execution_order) VALUES ('p','d1','power','{}',0)`).run();
  }
}
const counts = (db) => Object.fromEntries(['devices','device_states','capability_states','group_devices','command_history','preset_actions'].map((t)=>[t,db.prepare(`SELECT COUNT(*) n FROM ${t}`).get().n]));
const rebuild = `
PRAGMA foreign_keys = OFF;
CREATE TABLE devices_new AS SELECT 1 AS owner_id, * FROM devices;
DROP TABLE devices;
ALTER TABLE devices_new RENAME TO devices;
`;
for (const pinned of [false, true]) {
  const dir = fs.mkdtempSync(path.join(os.tmpdir(), 'm-'));
  for (const f of fs.readdirSync(REAL).filter(f=>f.endsWith('.sql'))) fs.copyFileSync(path.join(REAL,f), path.join(dir,f));
  const db = openDatabase(':memory:'); runMigrations(db, dir); seed(db, pinned);
  console.log(`rebuild test, device-pinned preset=${pinned}: before`, JSON.stringify(counts(db)), 'foreign_keys=', db.pragma('foreign_keys',{simple:true}));
  fs.writeFileSync(path.join(dir,'999_rebuild.sql'), rebuild);
  try { runMigrations(db, dir); console.log('  migration: applied. after', JSON.stringify(counts(db))); const e=db.prepare('SELECT message, device_id FROM activity_events').all(); console.log('  activity rows', JSON.stringify(e)); }
  catch (e) { console.log('  migration: FAILED:', e.message, ' after', JSON.stringify(counts(db))); }
}
